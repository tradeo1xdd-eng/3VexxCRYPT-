-- Función para cargar la configuración guardada
local function LoadConfig()
    if not isfile or not readfile then return end
    pcall(function()
        if isfile(ConfigFileName) then
            local decoded = HttpService:JSONDecode(readfile(ConfigFileName))
            if decoded and decoded.Settings then
                local s = decoded.Settings
                Settings.AimbotEnabled = s.AimbotEnabled or Settings.AimbotEnabled
                if s.AimKey and Enum.KeyCode[s.AimKey] then Settings.AimKey = Enum.KeyCode[s.AimKey] end
                Settings.AimPart = s.AimPart or Settings.AimPart
                Settings.VehicleAimPart = s.VehicleAimPart or Settings.VehicleAimPart
                Settings.FOV = s.FOV or Settings.FOV
                Settings.MaxDistance = s.MaxDistance or Settings.MaxDistance
                Settings.ShowFOV = s.ShowFOV ~= nil and s.ShowFOV or Settings.ShowFOV
                Settings.ShowTracer = s.ShowTracer ~= nil and s.ShowTracer or Settings.ShowTracer
                Settings.NoRecoil = s.NoRecoil ~= nil and s.NoRecoil or Settings.NoRecoil
                Settings.NoSpread = s.NoSpread ~= nil and s.NoSpread or Settings.NoSpread
                Settings.WallCheck = s.WallCheck ~= nil and s.WallCheck or Settings.WallCheck
            end
            if decoded and decoded.Visuals then
                local v = decoded.Visuals
                PlayerESP = v.PlayerESP ~= nil and v.PlayerESP or PlayerESP
                NameESP = v.NameESP ~= nil and v.NameESP or NameESP
                DistanceESP = v.DistanceESP ~= nil and v.DistanceESP or DistanceESP
                HealthESP = v.HealthESP ~= nil and v.HealthESP or HealthESP
                WeaponESP = v.WeaponESP ~= nil and v.WeaponESP or WeaponESP
                InventoryViewerEnabled = v.InventoryViewer ~= nil and v.InventoryViewer or InventoryViewerEnabled
                HideNameEnabled = v.HideName ~= nil and v.HideName or HideNameEnabled
                AutoDetectFriends = v.AutoDetectFriends ~= nil and v.AutoDetectFriends or AutoDetectFriends
            end
            if decoded and decoded.Combat then
                local c = decoded.Combat
                autoBloodActive = c.AutoBlood ~= nil and c.AutoBlood or autoBloodActive
                autoMinigameEnabled = c.AutoMinigame ~= nil and c.AutoMinigame or autoMinigameEnabled
                skipCrateEnabled = c.SkipCrate ~= nil and c.SkipCrate or skipCrateEnabled
            end
            if decoded and decoded.Extras then
                local e = decoded.Extras
                fpsBoostEnabled = e.FPSBoost ~= nil and e.FPSBoost or fpsBoostEnabled
                droppedESPEnabled = e.DroppedESP ~= nil and e.DroppedESP or droppedESPEnabled
                autoPickupEnabled = e.AutoPickup ~= nil and e.AutoPickup or autoPickupEnabled
            end
        end
    end)
end

    for _, tab in ipairs({VisualTab, AimTab, BloodTab, ServiciosTab}) do
        tab.BackgroundColor3 = (tab == activeTab) and Colors.Active or Colors.BGLight

local function CreateSlider(frame, name, min, max, defaultVal, callback)
    local f = Instance.new("Frame")
    f.Size = UDim2.new(1, 0, 0, 52); f.BackgroundColor3 = Colors.BGSecondary; f.BorderSizePixel = 0; f.Parent = frame
    Instance.new("UICorner", f).CornerRadius = UDim.new(0, 10)

    valLabel.Size = UDim2.new(0, 50, 0, 18); valLabel.Position = UDim2.new(1, -60, 0, 4)
    valLabel.BackgroundTransparency = 1; valLabel.Text = tostring(defaultVal); valLabel.TextColor3 = Colors.Accent
    valLabel.TextSize = 12; valLabel.Font = Enum.Font.GothamBold; valLabel.TextXAlignment = Enum.TextXAlignment.Right; valLabel.Parent = f
    
    local sliderBg = Instance.new("Frame")
    sliderBg.Size = UDim2.new(1, -28, 0, 8); sliderBg.Position = UDim2.new(0, 14, 0, 32)
    sliderBg.BackgroundColor3 = Colors.BGLight; sliderBg.BorderSizePixel = 0; sliderBg.Parent = f
    Instance.new("UICorner", sliderBg).CornerRadius = UDim.new(1, 0)
    
    local sliderFill = Instance.new("Frame")
    sliderFill.Size = UDim2.new(math.clamp((defaultVal - min) / (max - min), 0, 1), 0, 1, 0)
    sliderFill.BackgroundColor3 = Colors.Active; sliderFill.BorderSizePixel = 0; sliderFill.Parent = sliderBg
    Instance.new("UICorner", sliderFill).CornerRadius = UDim.new(1, 0)
    
    local dragging = false
    local function updateInput(input)
        local pos = math.clamp((input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + (max - min) * pos)
        sliderFill.Size = UDim2.new(pos, 0, 1, 0)
        valLabel.Text = tostring(val)
        if callback then callback(val) end
    end
    
    sliderBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateInput(input)
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateInput(input)
        end
    end)
    
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                dragging = false
                SaveConfig()
            end
        end
    end)
end

local Items = ReplicatedStorage:FindFirstChild("Items")
local WeaponRegistry = {}
local function registerItems(folder)
    if not folder then return end
    for _, tool in ipairs(folder:GetChildren()) do
        if tool:IsA("Tool") then
            local handle = tool:FindFirstChild("Handle")
            local displayName = tool:GetAttribute("DisplayName") or tool.Name
            local itemId = tool:GetAttribute("ItemId") or tool:GetAttribute("Id") or tool.Name
            local rarity = tool:GetAttribute("RarityName") or "Common"
            local imageId = tool:GetAttribute("ImageId") or "rbxassetid://7072725737"
            local key
            if handle then
                local mesh = handle:FindFirstChildOfClass("SpecialMesh")
                if mesh and mesh.MeshId ~= "" then key = mesh.MeshId..(mesh.TextureId or "").."_RARITY_"..rarity
                elseif handle:IsA("MeshPart") and handle.MeshId ~= "" then key = handle.MeshId..(handle.TextureID or "").."_RARITY_"..rarity end
            end
            if not key and itemId and itemId ~= "" and itemId ~= tool.Name then key = "ITEMID_"..itemId.."_RARITY_"..rarity end
            if not key then key = "NAME_"..displayName.."_"..tool.Name.."_RARITY_"..rarity end
            WeaponRegistry[key] = {Name = displayName, Rarity = rarity, ImageId = imageId, ToolName = tool.Name}
        end
    end
end
if Items then
    for _, folderName in ipairs({"gun", "melee", "throwable", "consumable", "farming", "misc", "rod", "fish"}) do
        local folder = Items:FindFirstChild(folderName)
        if folder then registerItems(folder) end
    end
    Items.ChildAdded:Connect(function(child) if child:IsA("Folder") then registerItems(child) end end)
end

local function getItemKey(tool)
    local handle = tool:FindFirstChild("Handle")
    local displayName = tool:GetAttribute("DisplayName") or tool.Name
    local itemId = tool:GetAttribute("ItemId") or tool:GetAttribute("Id") or tool.Name
    local rarity = tool:GetAttribute("RarityName") or "Common"
    if handle then
        local mesh = handle:FindFirstChildOfClass("SpecialMesh")
        if mesh and mesh.MeshId ~= "" then return mesh.MeshId..(mesh.TextureId or "").."_RARITY_"..rarity end
        if handle:IsA("MeshPart") and handle.MeshId ~= "" then return handle.MeshId..(handle.TextureID or "").."_RARITY_"..rarity end
    end
    if itemId and itemId ~= "" and itemId ~= tool.Name then return "ITEMID_"..itemId.."_RARITY_"..rarity end
    return "NAME_"..displayName.."_"..tool.Name.."_RARITY_"..rarity
end

local function getDisplayName(player)
    if player == LocalPlayer then
        return FAKE_NAME
    elseif ExcludedPlayers[player.Name] and PlayerFakeNames[player.Name] then
        return PlayerFakeNames[player.Name]
    else
        return player.Name
    end
end

local function assignFakeName(playerName)
    if not PlayerFakeNames[playerName] then
        local usedNames = {}
        for _, name in pairs(PlayerFakeNames) do usedNames[name] = true end
        for _, fakeName in ipairs(FriendFakeNames) do
            if not usedNames[fakeName] then
                PlayerFakeNames[playerName] = fakeName
                return fakeName
            end
        end
        PlayerFakeNames[playerName] = "Friend" .. math.random(1, 99)
    end
    return PlayerFakeNames[playerName]
end

local function ocultarNombreEnTexto(textoOriginal)
    if not HideNameEnabled or not textoOriginal then return textoOriginal or "" end
    local realName = LocalPlayer.Name
    local displayName = LocalPlayer.DisplayName
    local modificado = textoOriginal:gsub(realName, FAKE_NAME)
    modificado = modificado:gsub(displayName, FAKE_NAME)
    for playerName, fakeName in pairs(PlayerFakeNames) do
        modificado = modificado:gsub(playerName, fakeName)
    end
    return modificado
end

local function aplicarFiltroGlobal(objeto)
    if objeto:IsA("TextLabel") or objeto:IsA("TextBox") or objeto:IsA("TextButton") then
        objeto.Text = ocultarNombreEnTexto(objeto.Text)
        objeto:GetPropertyChangedSignal("Text"):Connect(function()
            if HideNameEnabled then
                local nuevo = ocultarNombreEnTexto(objeto.Text)
                if objeto.Text ~= nuevo then objeto.Text = nuevo end
            end
        end)
    end
end

            end

end

local function RefreshFriends()
    for _, v in pairs(FriendFrame:GetChildren()) do if v:IsA("TextButton") then v:Destroy() end end
    local sortedPlayers = Players:GetPlayers()
    table.sort(sortedPlayers, function(a, b) return a.Name:lower() < b.Name:lower() end)
    for _, plr in pairs(sortedPlayers) do
        if plr ~= LocalPlayer then
            local btn = Instance.new("TextButton")
            btn.Size = UDim2.new(1, -12, 0, 28); btn.Position = UDim2.new(0, 6, 0, 0)
            btn.BackgroundColor3 = ExcludedPlayers[plr.Name] and Color3.fromRGB(20, 50, 30) or Colors.BGSecondary

            local displayName = ExcludedPlayers[plr.Name] and (PlayerFakeNames[plr.Name] or plr.Name) or plr.Name
            btn.TextColor3 = ExcludedPlayers[plr.Name] and Colors.FriendColor or Colors.Text
            btn.Text = (ExcludedPlayers[plr.Name] and "✓ " or "") .. displayName
            btn.Font = Enum.Font.GothamMedium; btn.TextSize = 12; btn.Parent = FriendFrame
            Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)
            btn.MouseButton1Click:Connect(function()

                if ExcludedPlayers[plr.Name] then
                    assignFakeName(plr.Name)
                else
                    PlayerFakeNames[plr.Name] = nil
                end
                RefreshFriends()
            end)
        end
    end
    task.wait(); FriendFrame.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y)
end

local function GetCenter() return Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2) end

local function GetAimPosition(character)
    if not character then return nil end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local head = character:FindFirstChild("Head")
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    local upperTorso = character:FindFirstChild("UpperTorso")
    local torso = character:FindFirstChild("Torso")
    if not rootPart then return nil end

    local seatPart = humanoid and humanoid.SeatPart
    if seatPart then
        if Settings.VehicleAimPart == "Head" then return head and head.Position or (seatPart.Position + Vector3.new(0, 2.5, 0))
        elseif Settings.VehicleAimPart == "Neck" then return (head and upperTorso) and ((head.Position + upperTorso.Position) / 2) or (head and head.Position + Vector3.new(0, -0.5, 0)) or (seatPart.Position + Vector3.new(0, 2, 0))
        elseif Settings.VehicleAimPart == "Torso" then return upperTorso and upperTorso.Position or torso and torso.Position or (seatPart.Position + Vector3.new(0, 1.5, 0)) end
        return head and head.Position or (seatPart.Position + Vector3.new(0, 2.5, 0))
    end

    if Settings.AimPart == "DynamicHitbox" then
        if head and upperTorso then
            return (head.Position * 0.4) + (upperTorso.Position * 0.6) + Vector3.new(0, 0.2, 0)
        elseif head and torso then
            return (head.Position * 0.4) + (torso.Position * 0.6)
        end
        return head and head.Position or rootPart.Position
    elseif Settings.AimPart == "Head" then 
        return head and head.Position or rootPart.Position
    elseif Settings.AimPart == "Neck" then 
        return (head and upperTorso) and ((head.Position + upperTorso.Position) / 2) or rootPart.Position
    elseif Settings.AimPart == "Torso" then 
        return upperTorso and upperTorso.Position or torso and torso.Position or rootPart.Position
    end
    
    return head and head.Position or rootPart.Position
end

local function GetClosestPlayer()
    local Closest, ClosestDistance = nil, Settings.FOV
    local center = GetCenter()
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and not ExcludedPlayers[plr.Name] and plr.Character then
            local humanoid = plr.Character:FindFirstChildOfClass("Humanoid")
            local aimPos = GetAimPosition(plr.Character)
            if aimPos and humanoid and humanoid.Health > 0 and humanoid:GetState() ~= Enum.HumanoidStateType.Dead then
                local pos, visible = Camera:WorldToViewportPoint(aimPos)
                if visible then
                    local dist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                    local root = plr.Character:FindFirstChild("HumanoidRootPart")
                    local distFromPlayer = (root and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")) and (LocalPlayer.Character.HumanoidRootPart.Position - root.Position).Magnitude or 0
                    
                    if dist < ClosestDistance and distFromPlayer <= Settings.MaxDistance then
                        if Settings.WallCheck then
                            local rayParams = RaycastParams.new()
                            rayParams.FilterType = Enum.RaycastFilterType.Exclude
                            rayParams.FilterDescendantsInstances = {LocalPlayer.Character}
                            local ray = workspace:Raycast(Camera.CFrame.Position, (aimPos - Camera.CFrame.Position).Unit * 500, rayParams)
                            if not ray or ray.Instance:IsDescendantOf(plr.Character) then
                                ClosestDistance = dist; Closest = plr
                            end
                        else
                            ClosestDistance = dist; Closest = plr
                        end
                    end
                end
            end
        end
    end
    return Closest
end

local Counter
pcall(function()
    for _, v in ipairs(getgc(true)) do
        if typeof(v) == "table" and rawget(v, "event") and rawget(v, "func") then Counter = v; break end
    end
end)
local function netGet(...)
    if not Counter or not Counter.func then return end
    local args = {...}
    for i, v in ipairs(args) do
        if typeof(v) == "Instance" and v:IsA("Model") and #v:GetChildren() == 0 then
            local dropped = Workspace:FindFirstChild("DroppedItems")
            if dropped then
                local model = dropped:FindFirstChildWhichIsA("Model")
                if model then args[i] = model else return end
            else return end
        end
    end
    Counter.func = (Counter.func or 0) + 1
    local get = ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("Get")
    return get:InvokeServer(Counter.func, unpack(args))
end

        end
    end
end)
